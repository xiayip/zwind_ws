cmake_minimum_required(VERSION 3.16)
project(zephyr_surround_vision LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(image_transport REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d features2d)

add_library(surround_renderer SHARED src/pair_renderer.cpp)
target_compile_features(surround_renderer PUBLIC cxx_std_17)
target_compile_options(surround_renderer PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(surround_renderer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(surround_renderer PUBLIC yaml-cpp opencv_core opencv_imgproc opencv_imgcodecs opencv_features2d)

add_library(panorama_component SHARED src/panorama_node.cpp)
target_compile_features(panorama_component PUBLIC cxx_std_17)
target_compile_options(panorama_component PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(panorama_component PRIVATE surround_renderer
  rclcpp::rclcpp rclcpp_components::component image_transport::image_transport)
ament_target_dependencies(panorama_component PUBLIC sensor_msgs std_msgs rcl_interfaces)
rclcpp_components_register_nodes(panorama_component "zephyr_surround_vision::PanoramaNode")

add_executable(render_pair src/render_pair.cpp)
target_link_libraries(render_pair PRIVATE surround_renderer)
add_executable(surround_container src/component_container.cpp)
target_compile_features(surround_container PUBLIC cxx_std_17)
target_include_directories(surround_container PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(surround_container PRIVATE rclcpp::rclcpp rclcpp_components::component_manager opencv_core)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_renderer test/test_renderer.cpp)
  target_link_libraries(test_renderer surround_renderer)
  target_compile_definitions(test_renderer PRIVATE RIG_FILE="${CMAKE_CURRENT_SOURCE_DIR}/config/rig_20261028.yaml")
  add_executable(panorama_probe test/topic_probe.cpp)
  target_compile_features(panorama_probe PUBLIC cxx_std_17)
  target_link_libraries(panorama_probe PRIVATE rclcpp::rclcpp)
  ament_target_dependencies(panorama_probe PUBLIC sensor_msgs)
  install(TARGETS panorama_probe DESTINATION lib/${PROJECT_NAME})
  add_library(panorama_intra_probe SHARED test/intra_process_probe.cpp)
  target_compile_features(panorama_intra_probe PUBLIC cxx_std_17)
  target_link_libraries(panorama_intra_probe PRIVATE rclcpp::rclcpp rclcpp_components::component)
  ament_target_dependencies(panorama_intra_probe PUBLIC sensor_msgs std_msgs)
  rclcpp_components_register_nodes(panorama_intra_probe "zephyr_surround_vision::IntraProcessProbe")
  install(TARGETS panorama_intra_probe LIBRARY DESTINATION lib)
endif()
install(TARGETS surround_renderer panorama_component LIBRARY DESTINATION lib)
install(TARGETS render_pair surround_container DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch web DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
